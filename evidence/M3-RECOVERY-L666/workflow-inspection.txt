.github/workflows/aien-verification.yml
# AIEN Verification: the GitHub thin gate (ADR 0033 Decision 11, inventory contract C5).
#
# PHASE 3 (2026-10-18, Drake: "all testing on the local network; the Spark is the only test
# judge"). This is the only required check on main. The Spark worker (aien-test worker) runs
# the tests and signs a bundle; ~/.local/lib/aien-local-ci/pipeline.sh on the Spark publishes
# it to aien-receipts and reruns this workflow once it is there. A missing bundle fails this
# check (NOT_RUN) until the rerun. Pull requests labeled ready-to-merge are merged by the desk.
#
# What it does: checkout; read the trust anchors from the BASE ref (origin/main), never from
# the head under test (amendment A1); crumb verify; fetch <repo>/<sha>/bundle.json from
# aien-dev/aien-receipts; verify the bundle signature and content with the pinned aien-test
# build; run the repository's portable smoke set (gates whose requires is only "portable").
#
# GitHub-hosted compute: this workflow compiles only the two pinned verifier tools (the crumb
# checker, and aien-test once the repository is enrolled). It never builds or tests this
# repository's own code; the portable smoke set is text scans and format checks only.
#
# PHASE 3 rules (marked "PHASE3" below): a missing bundle is NOT_RUN (this check fails; the
# pipeline reruns it after publication), and a trust-surface change fails unless the verified
# bundle's signed plan shows the operator ran it by hand (plan.blocked == null).
name: AIEN Verification

on:
  pull_request:
  push:
    branches: [main]

permissions:
  contents: read

jobs:
  verify:
    name: AIEN Verification
    runs-on: ubuntu-latest
    timeout-minutes: 30
    env:
      REPO_SLUG: aien-dev/aien-sovereign-core
      AIEN_TEST_PIN: c26f4d7ac3156088469b27462390f5b08ef4f436
      RECEIPTS_RAW: https://raw.githubusercontent.com/aien-dev/aien-receipts/main
    steps:
      - name: Checkout the head under test
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Resolve the commit under test and the base ref
        id: ids
        run: |
          if [ "${{ github.event_name }}" = pull_request ]; then
            sha="${{ github.event.pull_request.head.sha }}"
          else
            sha="${{ github.sha }}"
          fi
          git fetch -q origin main
          echo "sha=$sha" >> "$GITHUB_OUTPUT"
          echo "base=$(git rev-parse origin/main)" >> "$GITHUB_OUTPUT"
          echo "commit under test: $sha"
          echo "base (trust anchors): $(git rev-parse origin/main)"

      - name: "Trust anchors from the base ref (A1: never from the head under test)"
        id: anchors
        run: |
          mkdir -p "$RUNNER_TEMP/anchors"
          base="${{ steps.ids.outputs.base }}"
          if git show "$base:.aien/verify.pub" > "$RUNNER_TEMP/anchors/verify.pub" 2>/dev/null; then
            echo "enrolled=true" >> "$GITHUB_OUTPUT"
            git show "$base:.aien/always" > "$RUNNER_TEMP/anchors/always" 2>/dev/null || : > "$RUNNER_TEMP/anchors/always"
            git show "$base:.aien/unmapped-ok" > "$RUNNER_TEMP/anchors/unmapped-ok" 2>/dev/null || : > "$RUNNER_TEMP/anchors/unmapped-ok"
            echo "verify.pub (base): $(cat "$RUNNER_TEMP/anchors/verify.pub")"
          else
            echo "enrolled=false" >> "$GITHUB_OUTPUT"
            # PHASE3: an enrolled repository whose base ref lost its key cannot verify anything.
            echo "::error::.aien/verify.pub is not on the base ref: no trusted key, nothing can be verified."
            exit 1
          fi
          # Trust-surface changes are run by the operator by hand (A1): BLOCKED_OPERATOR.
          changed=$(git diff --name-only "$base...${{ steps.ids.outputs.sha }}" -- .aien tests/gates .github/workflows .forgejo tools/aien-test || true)
          if [ -n "$changed" ]; then
            echo "blocked=true" >> "$GITHUB_OUTPUT"
            echo "::notice::BLOCKED_OPERATOR: this head changes the trust surface (the bundle must show an operator run):"; echo "$changed"
            # PHASE3: enforced in the verify step: the verified bundle's signed plan must say blocked == null.
          else
            echo "blocked=false" >> "$GITHUB_OUTPUT"
          fi

      - name: Checkout the pinned crumb tool source (aien-architecture @ 94fffbcf88492c5a1c54211b5126c8a6c370a98d)
        uses: actions/checkout@v4
        with:
          repository: aien-spark/aien-architecture
          ref: 94fffbcf88492c5a1c54211b5126c8a6c370a98d
          path: .crumb-tool-src

      - name: Build the crumb tool, then remove its source from this tree
        run: |
          make -C .crumb-tool-src/tools/crumb crumb
          mkdir -p "$RUNNER_TEMP/bin" && cp .crumb-tool-src/tools/crumb/crumb "$RUNNER_TEMP/bin/crumb"
          rm -rf .crumb-tool-src
          echo "$RUNNER_TEMP/bin" >> "$GITHUB_PATH"

      - name: Crumbs must be exactly what the compiler produces now (stays GitHub-hosted, D11)
        run: crumb verify .

      - name: Install Stable Rust Toolchain (for the pinned aien-test build and the smoke set)
        if: steps.anchors.outputs.enrolled == 'true'
        uses: dtolnay/rust-toolchain@stable
        with:
          components: rustfmt

      - name: Checkout the pinned aien-test source (aien-sovereign-core @ ${{ env.AIEN_TEST_PIN }})
        if: steps.anchors.outputs.enrolled == 'true'
        uses: actions/checkout@v4
        with:
          repository: aien-spark/aien-sovereign-core
          ref: ${{ env.AIEN_TEST_PIN }}
          path: .aien-test-src

      - name: Cache the pinned aien-test build
        if: steps.anchors.outputs.enrolled == 'true'
        uses: actions/cache@v4
        with:
          path: |
            ~/.cargo/registry
            ~/.cargo/git
            .aien-test-src/target
          key: aien-test-${{ runner.os }}-${{ env.AIEN_TEST_PIN }}

      - name: Build aien-test from the pin (never from the tree under test), then remove the source
        if: steps.anchors.outputs.enrolled == 'true'
        run: |
          cargo build --release -p aien-test --manifest-path .aien-test-src/Cargo.toml
          cp .aien-test-src/target/release/aien-test "$RUNNER_TEMP/bin/aien-test" 2>/dev/null || { mkdir -p "$RUNNER_TEMP/bin"; cp .aien-test-src/target/release/aien-test "$RUNNER_TEMP/bin/aien-test"; }
          echo "$RUNNER_TEMP/bin" >> "$GITHUB_PATH"
          rm -rf .aien-test-src
          git status --porcelain --untracked-files=normal

      - name: Fetch the signed verification bundle for this commit
        id: bundle
        run: |
          url="$RECEIPTS_RAW/aien-sovereign-core/${{ steps.ids.outputs.sha }}/bundle.json"
          echo "bundle url: $url"
          if curl -fsSL "$url" -o "$RUNNER_TEMP/bundle.json"; then
            echo "present=true" >> "$GITHUB_OUTPUT"
          else
            echo "present=false" >> "$GITHUB_OUTPUT"
            # PHASE3: NOT_RUN. The Spark pipeline reruns this workflow after it publishes the bundle.
            echo "::error::NOT_RUN: no signed Spark bundle for ${{ steps.ids.outputs.sha }} yet. The Spark pipeline reruns this check once the bundle is published."
            exit 2
          fi

      - name: Verify the bundle with the base-ref public key
        if: steps.bundle.outputs.present == 'true' && steps.anchors.outputs.enrolled == 'true'
        run: |
          required=$(grep -v '^#' "$RUNNER_TEMP/anchors/always" | grep -v '^$' | paste -sd, -)
          set +e
          aien-test verify-bundle "$RUNNER_TEMP/bundle.json" \
            --repo "github.com/$REPO_SLUG" \
            --commit "${{ steps.ids.outputs.sha }}" \
            --pubkey "$RUNNER_TEMP/anchors/verify.pub" \
            ${required:+--require "$required"}
          rc=$?
          set -e
          echo "verify-bundle exit $rc (0 PASS, 1 FAIL, 2 NOT_RUN, 3 REJECTED)"
          [ "$rc" = 0 ] || exit "$rc"
          # PHASE3: a trust-surface head passes only on an operator run: the plan member is inside
          # the signed bytes just verified, and only an operator run of such a head has blocked == null.
          if [ "${{ steps.anchors.outputs.blocked }}" = true ]; then
            if jq -e '.plan.blocked == null' "$RUNNER_TEMP/bundle.json" >/dev/null; then
              echo "trust-surface change: the signed bundle shows an operator run (plan.blocked == null)"
            else
              echo "::error::BLOCKED_OPERATOR: this head changes the trust surface and the bundle is not from an operator run"; exit 1
            fi
          fi

      - name: Portable smoke set (gates whose requires is only portable, named at the base ref)
        if: steps.anchors.outputs.enrolled == 'true' && steps.anchors.outputs.blocked == 'false'
        run: |
          base="${{ steps.ids.outputs.base }}"
          names=$(for f in $(git ls-tree -r --name-only "$base" -- tests/gates | grep '\.gate$'); do
            git show "$base:$f" | awk '
              /^gate:/ {g=$2}
              /^requires:/ {inr=1; next}
              inr && /^  - / { if ($2 != "portable") other=1; next }
              /^[a-z]/ {inr=0}
              END { if (g != "" && !other) print g }'
          done)
          echo "portable smoke set: $(echo $names)"
          mkdir -p "$RUNNER_TEMP/evidence"
          fail=0
          for g in $names; do
            echo "::group::$g"
            if timeout 600 aien-test run "$g" --evidence-dir "$RUNNER_TEMP/evidence"; then echo "$g: PASS"; else echo "::error::$g: not PASS"; fail=1; fi
            echo "::endgroup::"
          done
          exit $fail

      - name: "Trust-surface change: the portable smoke set is skipped (the operator ran this head on the Spark)"
        if: steps.anchors.outputs.blocked == 'true'
        run: |
          echo "AIEN Verification: trust-surface head, accepted on the verified operator-run bundle above."

name: Release Dry Run (no publish, no secrets)

# Runs the release gate checks offline on every pull request. It cannot publish: read-only permissions, no secrets,
# no signing, no upload. Guarded by scripts/test-dry-run-workflow.sh. Decision D1 (aien-architecture#156): no signing
# key is generated or stored now and nothing is published.
on:
  workflow_dispatch:

permissions:
  contents: read

jobs:
  gate-dry-run:
    name: Release gate dry run
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Source
        uses: actions/checkout@v4

      - name: Workflow cannot publish (static guard)
        run: bash scripts/test-dry-run-workflow.sh

      - name: Tag release gate stays fail-closed (static guard)
        run: bash scripts/test-release-gate-wiring.sh

      - name: No release key in CI; release.yml only drafts; publish only after the offline-signature gate (static guard)
        run: bash scripts/test-release-signing-workflow.sh

      # Report only: main may move ahead of the candidate's omega-commit. A release is cut from a commit whose tree
      # passes this gate; release.yml enforces it on tag push (guarded by scripts/test-release-gate-wiring.sh).
      - name: Tree gate on this revision (report, does not block)
        continue-on-error: true
        run: |
          if out="$(scripts/check-release-candidate.sh 2>&1)"; then
            echo "Release gate: this tree IS releasable as ${out#candidate=}" | tee -a "$GITHUB_STEP_SUMMARY"
          else
            echo "Release gate: this tree is NOT releasable as the candidate: $out" | tee -a "$GITHUB_STEP_SUMMARY"
            exit 1
          fi

      - name: Gate refuses non-candidate trees and packages (real CAND manifest, throwaway binaries)
        run: bash scripts/test-release-real-tree.sh

.github/workflows/release-publish.yml
name: "Release publish (stage 3: verify the offline signature, then publish the draft)"

# Offline-signing handoff, stage 3 (Drake ruling 1, sc#385; docs/release/RELEASE_SIGNING.md). Run by hand after the
# operator has uploaded SHA256SUMS.txt.sig, made offline with the public-release key, to the draft release that
# release.yml drafted for the tag. This workflow holds no key and signs nothing: it checks out the tagged commit (so the
# key it trusts is the one pinned in that commit), downloads the draft's assets, runs scripts/release-publish-gate.sh,
# and publishes only when the gate passes. Unsigned, mismatched, dry-run, internal-test-key and placeholder-key releases
# are refused. Static guard: scripts/test-release-signing-workflow.sh.
on:
  workflow_dispatch:
    inputs:
      tag:
        description: "The v* tag whose draft release now carries SHA256SUMS.txt.sig"
        required: true
        type: string

permissions:
  contents: write

jobs:
  verify-and-publish:
    name: Verify the offline signature and publish
    runs-on: ubuntu-latest
    env:
      TAG: ${{ inputs.tag }}
      GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
    steps:
      - name: Check the tag name
        run: |
          printf '%s\n' "$TAG" | grep -qxE 'v[0-9]+\.[0-9]+\.[0-9]+(-[0-9A-Za-z.]+)?' || { echo "not a release tag: $TAG" >&2; exit 1; }

      - name: Checkout the tagged commit (its docs/release/allowed_signers is the key this release is checked against)
        uses: actions/checkout@v4
        with:
          ref: refs/tags/${{ inputs.tag }}

      - name: Download the draft release assets
        run: |
          [ "$(gh release view "$TAG" --json isDraft --jq .isDraft)" = true ] || { echo "$TAG is not a draft release; nothing to publish" >&2; exit 1; }
          gh release download "$TAG" --dir release-assets
          ls -l release-assets

      - name: Publish gate (offline signature valid for the pinned release key, every asset matches, no dry-run asset)
        run: bash scripts/release-publish-gate.sh release-assets docs/release/allowed_signers

      - name: Publish the verified draft
        run: gh release edit "$TAG" --draft=false

.github/workflows/release.yml
name: Multi-OS Cross-Platform Release

on:
  push:
    tags:
      - "v*"
  workflow_dispatch:
    inputs:
      dry_run:
        description: "Dry run: build, package, sign with a throwaway key, verify, upload workflow artifacts only. Never creates a release."
        type: boolean
        default: true

permissions:
  contents: write

jobs:
  build-release:
    name: Build ${{ matrix.target }}
    runs-on: ${{ matrix.os }}
    strategy:
      fail-fast: false
      matrix:
        include:
          # Only the native candidate target is released: the candidate manifest has executable digests for it alone,
          # and the gate refuses any package without them (scripts/check-release-candidate.sh --native is required).
          - os: ubuntu-24.04-arm
            target: aarch64-unknown-linux-gnu
            artifact_name: sovereign-linux-aarch64.tar.gz
            use_tar: true
            native_engine: true

    steps:
      - name: Checkout Source
        uses: actions/checkout@v4

      - name: GNU tar (reproducible packages need it)
        if: runner.os == 'macOS'
        run: brew install gnu-tar

      - name: Install pinned Rust toolchain (rust-toolchain.toml)
        run: |
          rustup toolchain install "$(sed -n 's/^channel *= *"\(.*\)"/\1/p' rust-toolchain.toml)" --profile minimal
          rustc --version

      - name: Release gate (candidate named, omega.lock and Cargo.lock pins equal the candidate's)
        id: gate
        continue-on-error: ${{ github.event_name == 'workflow_dispatch' && inputs.dry_run }}
        run: scripts/check-release-candidate.sh

      - name: Read pins (omega.lock, then omega physics.lock)
        id: pins
        run: echo "omega=$(tr -d '[:space:]' < omega.lock)" >> "$GITHUB_OUTPUT"

      # The native engine is linked only on the candidate target (linux aarch64, the GB10 host class).
      # Other targets build the CPU stub, are labelled as such in the release notes, and are not the candidate.
      - name: Checkout omega at omega.lock
        if: matrix.native_engine
        uses: actions/checkout@v4
        with:
          repository: aien-spark/omega
          ref: ${{ steps.pins.outputs.omega }}
          path: omega

      - name: Read physics pin and checkout physics
        if: matrix.native_engine
        id: physpin
        run: |
          echo "physics=$(tr -d '[:space:]' < omega/physics.lock)" >> "$GITHUB_OUTPUT"
          echo "aienos=$(tr -d '[:space:]' < omega/aienos.lock)" >> "$GITHUB_OUTPUT"

      - name: Checkout physics at omega's physics.lock
        if: matrix.native_engine
        uses: actions/checkout@v4
        with:
          repository: aien-spark/physics
          ref: ${{ steps.physpin.outputs.physics }}
          path: physics

      # omega builds its aienos authority capability from the aienos commit pinned in omega/aienos.lock (as ci.yml does).
      - name: Checkout aienos at omega's aienos.lock
        if: matrix.native_engine
        uses: actions/checkout@v4
        with:
          repository: aien-spark/aienos
          ref: ${{ steps.physpin.outputs.aienos }}
          path: aienos

      - name: Build the candidate way (scripts/repro-build.sh, fresh CARGO_HOME, no mojo)
        env:
          CARGO_HOME: ${{ runner.temp }}/cargo-home
        run: |
          if command -v mojo >/dev/null 2>&1; then echo "mojo is on PATH; the candidate build runs without it" >&2; exit 1; fi
          if [ "${{ matrix.native_engine }}" = "true" ]; then
            export AIEN_OMEGA_DIR="$GITHUB_WORKSPACE/omega" AIEN_PHYSICS_DIR="$GITHUB_WORKSPACE/physics" AIEN_AIENOS_LOCK_REPO="$GITHUB_WORKSPACE/aienos"
          fi
          rm -rf "$CARGO_HOME"
          cargo fetch --locked
          # The one recipe (docs/release/BUILD_RECIPE.md): helpers as one group, then aien-cli alone and last.
          scripts/release-build.sh

      - name: Package Release Assets (reproducible tarball, release.toml inside)
        env:
          AIEN_DRY_RUN: ${{ github.event_name == 'workflow_dispatch' && inputs.dry_run && '1' || '' }}
        run: |
          bash scripts/package-release.sh "${{ matrix.artifact_name }}"
          AIEN_SHA="$(sha256sum target/release/aien-cli 2>/dev/null || shasum -a 256 target/release/aien-cli)"
          printf '%s  %s  native_engine=%s\n' "${AIEN_SHA%% *}" "${{ matrix.artifact_name }}" "${{ matrix.native_engine }}" > "aien-cli-sha256-${{ matrix.target }}.txt"
          cat "aien-cli-sha256-${{ matrix.target }}.txt"

      - name: Check the package against the candidate (model inputs, files, aien-cli and helper digests, all bound to the candidate manifest)
        id: pkgcheck
        continue-on-error: ${{ github.event_name == 'workflow_dispatch' && inputs.dry_run }}
        run: |
          scripts/check-release-candidate.sh --package "${{ matrix.artifact_name }}" --native

      - name: Record dry-run status
        if: always() && github.event_name == 'workflow_dispatch' && inputs.dry_run
        run: |
          {
            echo "DRY RUN. This is not a release and must not be installed as a candidate."
            echo "release gate: ${{ steps.gate.outcome }}"
            echo "package check: ${{ steps.pkgcheck.outcome }}"
          } > "DRY-RUN-STATUS-${{ matrix.target }}.txt"
          cat "DRY-RUN-STATUS-${{ matrix.target }}.txt"

      - name: Upload Artifact
        uses: actions/upload-artifact@v4
        with:
          name: ${{ matrix.target }}
          path: |
            ${{ matrix.artifact_name }}
            aien-cli-sha256-${{ matrix.target }}.txt
            DRY-RUN-STATUS-${{ matrix.target }}.txt

  # Offline-signing handoff (Drake ruling 1, sc#385; docs/release/RELEASE_SIGNING.md). No private release key is ever
  # in CI, a secret, or this repository. Stage 1 (draft-release, below) builds and publishes checksums to a DRAFT
  # release. Stage 2 is the operator, offline: sign SHA256SUMS.txt and upload SHA256SUMS.txt.sig to the draft.
  # Stage 3 (.github/workflows/release-publish.yml) verifies that signature against docs/release/allowed_signers and
  # only then publishes. Static guard: scripts/test-release-signing-workflow.sh.
  dry-run-sign:
    name: "Dry run: label, sign with a throwaway key made in this job, verify (signature_check dry-run-key)"
    needs: build-release
    if: github.event_name == 'workflow_dispatch' && inputs.dry_run
    runs-on: ubuntu-latest
    permissions:
      contents: read
    steps:
      - name: Checkout Source
        uses: actions/checkout@v4

      - name: Download Build Artifacts
        uses: actions/download-artifact@v4
        with:
          path: release-artifacts
          merge-multiple: true

      - name: "Label, checksums, throwaway signature, verification (DRY RUN, not a release)"
        working-directory: release-artifacts
        run: |
          for a in sovereign-*.tar.gz; do
            tar -xzOf "$a" ./release.toml | grep -qx 'dry-run = true' || { echo "$a: release.toml is not labelled dry-run" >&2; exit 1; }
          done
          echo "DRY RUN: signed with a throwaway key generated in this job and deleted (signature_check dry-run-key). Not the AIEN release key. Not a release. Never publish." > DRY-RUN-NOT-A-RELEASE.txt
          cat aien-cli-sha256-*.txt > aien-cli-digests.txt
          sha256sum sovereign-* aien-cli-digests.txt DRY-RUN-*.txt > SHA256SUMS.txt
          umask 077
          ssh-keygen -q -t ed25519 -N "" -C aien-dry-run-throwaway -f "$RUNNER_TEMP/dry_key"
          printf "aien-release %s\n" "$(cut -d" " -f1,2 "$RUNNER_TEMP/dry_key.pub")" > dry-run-signers
          ssh-keygen -Y sign -f "$RUNNER_TEMP/dry_key" -n aien-release SHA256SUMS.txt
          rm -f "$RUNNER_TEMP/dry_key" "$RUNNER_TEMP/dry_key.pub"
          ../scripts/verify-release-assets.sh . dry-run-signers
          cat SHA256SUMS.txt

      - name: "Negative control: the publish gate refuses this dry run, under the pinned key and under its own key"
        run: |
          if bash scripts/release-publish-gate.sh release-artifacts docs/release/allowed_signers; then
            echo "publish gate accepted a dry run under docs/release/allowed_signers" >&2; exit 1
          fi
          if bash scripts/release-publish-gate.sh release-artifacts release-artifacts/dry-run-signers; then
            echo "publish gate accepted a dry run under its own throwaway key" >&2; exit 1
          fi
          echo "publish gate refused the dry run, as it must"

      - name: Upload dry-run artifacts (workflow artifacts only, no release)
        uses: actions/upload-artifact@v4
        with:
          name: dry-run-signed-assets
          path: release-artifacts

  # Stage 1. Never signs and never publishes: it creates a DRAFT release holding the archives and SHA256SUMS.txt.
  draft-release:
    name: "Stage 1: checksums and a draft release (unsigned; the operator signs offline)"
    needs: build-release
    if: github.event_name == 'push' || inputs.dry_run == false
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Source
        uses: actions/checkout@v4

      - name: Refuse anything but a v* tag
        run: |
          if [ "$GITHUB_REF_TYPE" != tag ] || [ "${GITHUB_REF_NAME#v}" = "$GITHUB_REF_NAME" ]; then
            echo "A release is drafted only from a v* tag (this run is on $GITHUB_REF_TYPE $GITHUB_REF_NAME). Nothing was drafted." >&2
            exit 1
          fi

      - name: Download Build Artifacts
        uses: actions/download-artifact@v4
        with:
          path: release-artifacts
          merge-multiple: true

      - name: Generate Checksums
        working-directory: release-artifacts
        run: |
          cat aien-cli-sha256-*.txt > aien-cli-digests.txt
          sha256sum sovereign-* aien-cli-digests.txt > SHA256SUMS.txt
          cat SHA256SUMS.txt

      - name: Create the draft release (no signature; not public until release-publish.yml verifies the offline signature)
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          TAG: ${{ github.ref_name }}
        run: |
          if gh release view "$TAG" >/dev/null 2>&1; then
            echo "A release named $TAG already exists. Refusing to replace it." >&2
            exit 1
          fi
          {
            echo "> DRAFT, AWAITING THE OFFLINE SIGNATURE. Not published. See docs/release/RELEASE_SIGNING.md."
            echo
            if [ -f RELEASE_NOTES.md ]; then cat RELEASE_NOTES.md; else echo "# AIEN Sovereign Core ${TAG}"; fi
            echo
            echo "## aien-cli digests (compare with the candidate manifest [executables] aien-cli-native-release)"
            echo
            echo "Columns: sha256 of target/release/aien-cli, archive, whether the native engine (omega at omega.lock) is linked. Only the native_engine=true linux-aarch64 row is the candidate binary."
            echo
            echo '```'
            cat release-artifacts/aien-cli-digests.txt
            echo '```'
          } > release-notes-with-digests.md
          cd release-artifacts
          gh release create "$TAG" --draft --verify-tag sovereign-* aien-cli-digests.txt SHA256SUMS.txt \
            --title "AIEN Sovereign Core ${TAG}" \
            --notes-file ../release-notes-with-digests.md
